{{- /* Render hook for markdown links. Hugo does not make plain markdown links language-aware, so a link written as `/docs` resolves to the default language on every translation. Root-relative destinations are therefore passed through relLangURL. Fragments, external URLs and other schemes are left alone. */ -}} {{- $dest := .Destination -}} {{- $external := false -}} {{- if or (strings.HasPrefix $dest "#") (strings.HasPrefix $dest "//") -}} {{- /* fragment or protocol-relative: leave untouched */ -}} {{- if strings.HasPrefix $dest "//" }}{{ $external = true }}{{ end -}} {{- else if strings.Contains $dest ":" -}} {{- /* has a scheme: http, https, mailto, tel */ -}} {{- $external = true -}} {{- else if strings.HasPrefix $dest "/" -}} {{- $dest = $dest | relLangURL -}} {{- end -}} {{ .Text }}{{- "" -}}
